    <script>

        async function sharePage() {

            const shareData = {

                title: "@om4r.64",

                text:
                    "Check out my links 🚀",

                url:
                    window.location.href
            };


            try {

                if (navigator.share) {

                    await navigator.share(
                        shareData
                    );

                } else {

                    await navigator.clipboard.writeText(
                        window.location.href
                    );

                    alert(
                        "Link copied!"
                    );
                }

            } catch (error) {

                console.log(
                    "Share cancelled."
                );
            }

        }

    </script>
